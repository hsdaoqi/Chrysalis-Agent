发现 3 处违规写入。重构方案：
  • Redis session store，token AES-256 加密存服务端
  • Cookie 仅放 session_id (httpOnly + secure + SameSite=Strict)
  • TTL 3600s，sliding window 续期
class RedisSessionStore: def create(self, user_id, token): sid = secrets.token_urlsafe(32) encrypted = self._encrypt(token) self.redis.setex(f"sess:{sid}", 3600, encrypted) return sid def validate(self, session_id): raw = self.redis.get(f"sess:{session_id}") if not raw: return None self.redis.expire(f"sess:{session_id}", 3600) # sliding window return self._decrypt(raw)
